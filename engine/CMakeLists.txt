cmake_minimum_required(VERSION 3.24)
set(CMAKE_SYSTEM_PROCESSOR AMD64)
project(suno_engine LANGUAGES C CXX)
set(CMAKE_SYSTEM_PROCESSOR AMD64)
set(CMAKE_CXX_STANDARD 23)
set(MUSCRIPTOR_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(MUSCRIPTOR_BUILD_BENCH OFF CACHE BOOL "" FORCE)
set(MUSCRIPTOR_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../vendor/muscriptor.cpp" CACHE PATH "muscriptor.cpp checkout")
add_subdirectory("${MUSCRIPTOR_SOURCE_DIR}/cpp" muscriptor)
if(TARGET ggml-vulkan)
  set_source_files_properties("${FETCHCONTENT_SOURCE_DIR_GGML}/src/ggml-vulkan/ggml-vulkan.cpp" TARGET_DIRECTORY ggml-vulkan PROPERTIES COMPILE_OPTIONS "-O1")
endif()
add_executable(suno_engine main.cpp)
target_link_libraries(suno_engine PRIVATE muscriptor_ggml)
if(MINGW)
  target_link_options(suno_engine PRIVATE -municode)
endif()
